export function getUpgradesForRecipe(recipe) {
  if (!recipe) return [];
  const text = (recipe.title + " " + (recipe.tags || []).join(" ")).toLowerCase();
  for (const rule of DOCTOR_RULES) {
    if (rule.match.some(k => text.includes(k))) return rule.upgrades;
  }
  return [];
}

export function applyUpgrade(recipe, upgrade) {
  if (!recipe || !upgrade) return null;

  const baseIngredients = recipe.ingredients || [];
  const baseSteps = recipe.steps || [];

  const remixedIngredients = [
    ...baseIngredients,
    "",
    "— Doctor It Up Adds —",
    ...upgrade.addIngredients
  ];

  const remixedSteps = [
    ...baseSteps,
    "",
    "— Doctor It Up Steps —",
    ...upgrade.stepAdds
  ];

  return {
    title: `${recipe.title} + ${upgrade.title}`,
    why: upgrade.why,
    ingredients: remixedIngredients,
    steps: remixedSteps
  };
}

cat > src/App.jsx <<'JSX'
import React, { useMemo, useState } from "react";
import { searchRecipes } from "./data/recipes.js";
import { applyUpgrade, getUpgradesForRecipe } from "./data/rules.js";

function Pill({ children }) {
  return <span className="pill">{children}</span>;
}

function ListBlock({ title, items }) {
  return (
    <div className="block">
      <h3>{title}</h3>
      <ul>
        {items.map((x, i) => (
          <li key={i}>{x}</li>
        ))}
      </ul>
    </div>
  );
}

export default function App() {
  const [query, setQuery] = useState("box cake");
  const [selectedRecipe, setSelectedRecipe] = useState(null);
  const [selectedUpgrade, setSelectedUpgrade] = useState(null);

  const results = useMemo(() => searchRecipes(query), [query]);

  const upgrades = useMemo(() => getUpgradesForRecipe(selectedRecipe), [selectedRecipe]);

  const remixed = useMemo(() => {
    if (!selectedRecipe || !selectedUpgrade) return null;
    return applyUpgrade(selectedRecipe, selectedUpgrade);
  }, [selectedRecipe, selectedUpgrade]);

  return (
    <div className="container">
      <div className="header">
        <div className="brand">
          <h1>Doctor It Up</h1>
          <p>Search a base recipe, then tap upgrades to remix it.</p>
        </div>
        <div className="pillGroup">
          <Pill>MVP</Pill>
          <Pill>Capacitor-ready</Pill>
        </div>
      </div>

      <div className="card" style={{ marginTop: 14 }}>
        <div className="searchBox">
          <input
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder='Try: "box cake", "brownies", "ramen"'
            aria-label="Search recipes"
          />
          <button
            className="btn"
            onClick={() => {
              // pick first result for convenience
              const first = searchRecipes(query)[0] || null;
              setSelectedRecipe(first);
              setSelectedUpgrade(null);
            }}
          >
            Search
          </button>
        </div>

        <div className="hint">
          Tip: click “Search” then choose a recipe from results.
        </div>
      </div>

      <div className="row" style={{ marginTop: 14 }}>
        <div className="col">
          <div className="card">
            <h2>Results</h2>
            {results.length === 0 ? (
              <p className="muted">No results yet. Try “box cake”, “brownies”, or “ramen”.</p>
            ) : (
              <div className="list">
                {results.map((r) => (
                  <button
                    key={r.id}
                    className={"listItem " + (selectedRecipe?.id === r.id ? "active" : "")}
                    onClick={() => {
                      setSelectedRecipe(r);
                      setSelectedUpgrade(null);
                    }}
                  >
                    <div className="listItemTitle">{r.title}</div>
                    <div className="listItemTags">
                      {r.tags.slice(0, 3).map((t) => (
                        <span key={t} className="tag">{t}</span>
                      ))}
                    </div>
                  </button>
                ))}
              </div>
            )}
          </div>

          <div className="card" style={{ marginTop: 14 }}>
            <h2>Doctor It Up</h2>
            {!selectedRecipe ? (
              <p className="muted">Select a recipe first.</p>
            ) : upgrades.length === 0 ? (
              <p className="muted">No upgrades found for this recipe yet.</p>
            ) : (
              <div className="list">
                {upgrades.map((u) => (
                  <button
                    key={u.id}
                    className={"listItem " + (selectedUpgrade?.id === u.id ? "active" : "")}
                    onClick={() => setSelectedUpgrade(u)}
                  >
                    <div className="listItemTitle">{u.title}</div>
                    <div className="listItemSub muted">{u.why}</div>
                  </button>
                ))}
              </div>
            )}
          </div>
        </div>

        <div className="col">
          <div className="card">
            <h2>Base Recipe</h2>
            {!selectedRecipe ? (
              <p className="muted">Pick a recipe from the results.</p>
            ) : (
              <>
                <h3 className="recipeTitle">{selectedRecipe.title}</h3>
                <ListBlock title="Ingredients" items={selectedRecipe.ingredients} />
                <ListBlock title="Steps" items={selectedRecipe.steps} />
              </>
            )}
          </div>

          <div className="card" style={{ marginTop: 14 }}>
            <h2>Remixed Recipe</h2>
            {!remixed ? (
              <p className="muted">Choose an upgrade to generate a remixed version.</p>
            ) : (
              <>
                <h3 className="recipeTitle">{remixed.title}</h3>
                <p className="muted"><strong>Why:</strong> {remixed.why}</p>
                <ListBlock title="Ingredients" items={remixed.ingredients} />
                <ListBlock title="Steps" items={remixed.steps} />
              </>
            )}
          </div>
        </div>
      </div>

      <div className="bottomBar">
        <button
          className="btnWide"
          onClick={() => {
            alert(
              "Next step for iOS/TestFlight:\n\n1) Push this repo to GitHub\n2) Run: npm i\n3) Run: npm run cap:init\n4) Run: npm run cap:add:ios\n5) Run: npm run cap:sync\nThen open ios project in Xcode and upload to TestFlight.\n\n(Or automate with Codemagic.)"
            );
          }}
        >
          iOS / TestFlight Steps
        </button>
      </div>
    </div>
  );
}
